function renderSolutions() {
  const q = document.getElementById("solutionSearch").value.trim().toLowerCase();
  const book = document.getElementById("solutionBook").value;
  const result = solutions.filter(s => (!book || s.book === book) && `${s.book} ${s.page} ${s.result}`.toLowerCase().includes(q));
  const grid = document.getElementById("solutionsGrid");

  grid.innerHTML = result.map(s => `<article class="card">
    <div class="form-note">${escapeHTML(s.book)}</div>
    <h3 style="margin:5px 0 11px">${escapeHTML(s.page)}</h3>
    <div class="spoiler" data-result="${escapeHTML(s.result)}" data-image="${escapeHTML(s.image || "")}">
      <strong>Cliquer pour révéler</strong>
    </div>
  </article>`).join("") || `<div class="empty">Aucune solution trouvée.</div>`;

  grid.querySelectorAll(".spoiler").forEach(spoiler => spoiler.addEventListener("click",(event) => {
    if (event.target.closest(".solution-full-link")) return;

    spoiler.classList.toggle("revealed");
    if (!spoiler.classList.contains("revealed")) {
      spoiler.innerHTML = `<strong>Cliquer pour révéler</strong>`;
      return;
    }

    const image = spoiler.dataset.image;
    if (image) {
      spoiler.innerHTML = `
        <div class="solution-reveal">
          <img src="${image}" alt="${escapeHTML(spoiler.dataset.result)}" loading="lazy" referrerpolicy="no-referrer">
          <div class="solution-caption">
            <strong>${escapeHTML(spoiler.dataset.result)}</strong>
            <a class="solution-full-link" href="${image}" target="_blank" rel="noopener noreferrer">Voir en grand ↗</a>
          </div>
        </div>`;
    } else {
      spoiler.innerHTML = `<strong>${escapeHTML(spoiler.dataset.result)}</strong>`;
    }
  }));
}
document.getElementById("solutionSearch").addEventListener("input",renderSolutions);

function colorDistance(a,b) {
  const dr=a[0]-b[0],dg=a[1]-b[1],db=a[2]-b[2];
  return .30*dr*dr+.59*dg*dg+.11*db*db;
}
function avg(points) {
  let r=0,g=0,b=0;
  points.forEach(p=>{r+=p[0];g+=p[1];b+=p[2];});
  const n=Math.max(1,points.length);
  return [Math.round(r/n),Math.round(g/n),Math.round(b/n)];
}
function range(points,c) {
  let mn=255,mx=0;
  points.forEach(p=>{mn=Math.min(mn,p[c]);mx=Math.max(mx,p[c]);});
  return mx-mn;
}
function medianCut(pixels,count) {
  let boxes=[pixels.slice()];
  while(boxes.length<count) {
    let best=-1,score=-1,ch=0;
    boxes.forEach((box,i)=>{
      if(box.length<2)return;
      const rs=[0,1,2].map(c=>range(box,c));
      const cc=rs.indexOf(Math.max(...rs));
      const s=Math.max(...rs)*Math.sqrt(box.length);
      if(s>score){score=s;best=i;ch=cc;}
    });
    if(best<0)break;
    const box=boxes.splice(best,1)[0].sort((a,b)=>a[ch]-b[ch]);
    const mid=Math.floor(box.length/2);
    boxes.push(box.slice(0,mid),box.slice(mid));
  }
  return boxes.map(avg);
}
